import React, { useState, useRef, useEffect } from 'react';
import {
  View,
  Text,
  Modal,
  TouchableOpacity,
  StyleSheet,
  Alert,
  ActivityIndicator,
  Animated,
  Dimensions,
  Platform,
} from 'react-native';
import { LinearGradient } from 'expo-linear-gradient';
import { Ionicons } from '@expo/vector-icons';
import * as Location from 'expo-location';
import * as FileSystem from 'expo-file-system';
import { Camera } from 'expo-camera';
import { Audio } from 'expo-av';
import * as MediaLibrary from 'expo-media-library';
import AsyncStorage from '@react-native-async-storage/async-storage';

const { width, height } = Dimensions.get('window');

const PanicSystem = ({ visible, onClose }) => {
  // State Management
  const [isActivated, setIsActivated] = useState(false);
  const [isLoading, setIsLoading] = useState(false);
  const [loadingMessage, setLoadingMessage] = useState('');
  const [countdown, setCountdown] = useState(5);
  const [showCountdown, setShowCountdown] = useState(false);
  
  // Panic Status States
  const [locationTracking, setLocationTracking] = useState(false);
  const [imageCapturing, setImageCapturing] = useState(false);
  const [audioRecording, setAudioRecording] = useState(false);
  
  // Refs
  const cameraRef = useRef(null);
  const recordingRef = useRef(null);
  const locationIntervalRef = useRef(null);
  const imageIntervalRef = useRef(null);
  const countdownIntervalRef = useRef(null);
  
  // Animation
  const modalScaleAnim = useRef(new Animated.Value(0)).current;
  const pulseAnim = useRef(new Animated.Value(1)).current;

  // Permissions State
  const [permissions, setPermissions] = useState({
    location: false,
    camera: false,
    audio: false,
    mediaLibrary: false,
  });

  // Initialize animations and permissions
  useEffect(() => {
    if (visible) {
      // Animate modal appearance
      Animated.spring(modalScaleAnim, {
        toValue: 1,
        tension: 100,
        friction: 8,
        useNativeDriver: true,
      }).start();

      // Request all permissions when modal opens
      requestAllPermissions();
    } else {
      modalScaleAnim.setValue(0);
    }
  }, [visible]);

  // Pulse animation for panic button
  useEffect(() => {
    const pulseAnimation = Animated.loop(
      Animated.sequence([
        Animated.timing(pulseAnim, {
          toValue: 1.1,
          duration: 1000,
          useNativeDriver: true,
        }),
        Animated.timing(pulseAnim, {
          toValue: 1,
          duration: 1000,
          useNativeDriver: true,
        }),
      ])
    );
    
    if (visible && !isActivated) {
      pulseAnimation.start();
    } else {
      pulseAnimation.stop();
      pulseAnim.setValue(1);
    }

    return () => pulseAnimation.stop();
  }, [visible, isActivated]);

  // Request all necessary permissions
  const requestAllPermissions = async () => {
    try {
      setIsLoading(true);
      setLoadingMessage('Requesting permissions...');

      // Location permission
      const locationPermission = await Location.requestForegroundPermissionsAsync();
      
      // Camera permission
      const cameraPermission = await Camera.requestCameraPermissionsAsync();
      
      // Audio permission
      const audioPermission = await Audio.requestPermissionsAsync();
      
      // Media Library permission
      const mediaPermission = await MediaLibrary.requestPermissionsAsync();

      setPermissions({
        location: locationPermission.status === 'granted',
        camera: cameraPermission.status === 'granted',
        audio: audioPermission.status === 'granted',
        mediaLibrary: mediaPermission.status === 'granted',
      });

      setIsLoading(false);
      setLoadingMessage('');
    } catch (error) {
      console.error('Error requesting permissions:', error);
      setIsLoading(false);
      setLoadingMessage('');
      Alert.alert('Permission Error', 'Failed to request permissions. Please enable them in settings.');
    }
  };

  // Create directory for panic files
  const createPanicDirectory = async () => {
    try {
      const panicDir = `${FileSystem.documentDirectory}panic_files/`;
      const dirInfo = await FileSystem.getInfoAsync(panicDir);
      
      if (!dirInfo.exists) {
        await FileSystem.makeDirectoryAsync(panicDir, { intermediates: true });
      }
      
      return panicDir;
    } catch (error) {
      console.error('Error creating panic directory:', error);
      return FileSystem.documentDirectory;
    }
  };

  // Get current location
  const getCurrentLocation = async () => {
    try {
      if (!permissions.location) {
        throw new Error('Location permission not granted');
      }

      const location = await Location.getCurrentPositionAsync({
        accuracy: Location.Accuracy.High,
      });

      return {
        latitude: location.coords.latitude,
        longitude: location.coords.longitude,
        timestamp: new Date().toISOString(),
      };
    } catch (error) {
      console.error('Error getting location:', error);
      // Return fallback location (Mumbai, India based on user location)
      return {
        latitude: 19.0760,
        longitude: 72.8777,
        timestamp: new Date().toISOString(),
        error: 'Could not get precise location',
      };
    }
  };

  // Save location data to file
  const saveLocationData = async (locationData) => {
    try {
      const panicDir = await createPanicDirectory();
      const timestamp = new Date().toISOString().replace(/[:.]/g, '-');
      const fileName = `location_${timestamp}.json`;
      const filePath = `${panicDir}${fileName}`;

      await FileSystem.writeAsStringAsync(filePath, JSON.stringify(locationData, null, 2));
      console.log('Location saved to:', filePath);
      
      return filePath;
    } catch (error) {
      console.error('Error saving location:', error);
    }
  };

  // Start continuous location tracking
  const startLocationTracking = async () => {
    try {
      if (!permissions.location) {
        Alert.alert('Permission Required', 'Location permission is required for panic mode.');
        return;
      }

      setLocationTracking(true);
      console.log('Starting continuous location tracking...');

      // Get initial location
      const initialLocation = await getCurrentLocation();
      await saveLocationData([initialLocation]);

      // Create location tracking array
      const locationData = [initialLocation];

      // Track location every 10 seconds
      locationIntervalRef.current = setInterval(async () => {
        try {
          const currentLocation = await getCurrentLocation();
          locationData.push(currentLocation);
          
          // Save updated location data
          await saveLocationData(locationData);
          
          console.log('Location updated:', currentLocation);
        } catch (error) {
          console.error('Error in location tracking interval:', error);
        }
      }, 10000); // Every 10 seconds

    } catch (error) {
      console.error('Error starting location tracking:', error);
      setLocationTracking(false);
    }
  };

  // Stop location tracking
  const stopLocationTracking = () => {
    if (locationIntervalRef.current) {
      clearInterval(locationIntervalRef.current);
      locationIntervalRef.current = null;
    }
    setLocationTracking(false);
    console.log('Location tracking stopped');
  };

  // Capture photo
  const capturePhoto = async () => {
    try {
      if (!permissions.camera || !cameraRef.current) {
        console.log('Camera not available or permission not granted');
        return null;
      }

      const panicDir = await createPanicDirectory();
      const timestamp = new Date().toISOString().replace(/[:.]/g, '-');
      const fileName = `panic_photo_${timestamp}.jpg`;
      const filePath = `${panicDir}${fileName}`;

      const photo = await cameraRef.current.takePictureAsync({
        quality: 0.8,
        base64: false,
        skipProcessing: true,
      });

      // Move the photo to panic directory
      await FileSystem.moveAsync({
        from: photo.uri,
        to: filePath,
      });

      console.log('Photo captured and saved to:', filePath);
      return filePath;
    } catch (error) {
      console.error('Error capturing photo:', error);
      return null;
    }
  };

  // Start automatic image capturing
  const startImageCapturing = async () => {
    try {
      if (!permissions.camera) {
        Alert.alert('Permission Required', 'Camera permission is required for panic mode.');
        return;
      }

      setImageCapturing(true);
      console.log('Starting automatic image capturing...');

      // Capture initial photo
      await capturePhoto();

      // Capture photo every 15 seconds
      imageIntervalRef.current = setInterval(async () => {
        try {
          await capturePhoto();
        } catch (error) {
          console.error('Error in image capturing interval:', error);
        }
      }, 15000); // Every 15 seconds

    } catch (error) {
      console.error('Error starting image capturing:', error);
      setImageCapturing(false);
    }
  };

  // Stop image capturing
  const stopImageCapturing = () => {
    if (imageIntervalRef.current) {
      clearInterval(imageIntervalRef.current);
      imageIntervalRef.current = null;
    }
    setImageCapturing(false);
    console.log('Image capturing stopped');
  };

  // Start audio recording
  const startAudioRecording = async () => {
    try {
      if (!permissions.audio) {
        Alert.alert('Permission Required', 'Audio permission is required for panic mode.');
        return;
      }

      setAudioRecording(true);
      console.log('Starting audio recording...');

      // Configure audio mode
      await Audio.setAudioModeAsync({
        allowsRecordingIOS: true,
        playsInSilentModeIOS: true,
      });

      const panicDir = await createPanicDirectory();
      const timestamp = new Date().toISOString().replace(/[:.]/g, '-');
      const fileName = `panic_audio_${timestamp}.m4a`;
      const filePath = `${panicDir}${fileName}`;

      // Start recording
      const { recording } = await Audio.Recording.createAsync({
        android: {
          extension: '.m4a',
          outputFormat: Audio.RECORDING_OPTION_ANDROID_OUTPUT_FORMAT_MPEG_4,
          audioEncoder: Audio.RECORDING_OPTION_ANDROID_AUDIO_ENCODER_AAC,
          sampleRate: 44100,
          numberOfChannels: 2,
          bitRate: 128000,
        },
        ios: {
          extension: '.m4a',
          outputFormat: Audio.RECORDING_OPTION_IOS_OUTPUT_FORMAT_MPEG4AAC,
          audioQuality: Audio.RECORDING_OPTION_IOS_AUDIO_QUALITY_HIGH,
          sampleRate: 44100,
          numberOfChannels: 2,
          bitRate: 128000,
          linearPCMBitDepth: 16,
          linearPCMIsBigEndian: false,
          linearPCMIsFloat: false,
        },
        web: {
          mimeType: 'audio/webm',
          bitsPerSecond: 128000,
        },
      });

      recordingRef.current = recording;

      // Stop recording after 30 seconds
      setTimeout(async () => {
        try {
          if (recordingRef.current) {
            await recordingRef.current.stopAndUnloadAsync();
            const uri = recordingRef.current.getURI();
            
            // Move recording to panic directory
            if (uri) {
              await FileSystem.moveAsync({
                from: uri,
                to: filePath,
              });
              console.log('Audio recorded and saved to:', filePath);
            }
            
            recordingRef.current = null;
            setAudioRecording(false);
          }
        } catch (error) {
          console.error('Error stopping audio recording:', error);
          setAudioRecording(false);
        }
      }, 30000); // 30 seconds

    } catch (error) {
      console.error('Error starting audio recording:', error);
      setAudioRecording(false);
    }
  };

  // Stop audio recording
  const stopAudioRecording = async () => {
    try {
      if (recordingRef.current) {
        await recordingRef.current.stopAndUnloadAsync();
        recordingRef.current = null;
      }
      setAudioRecording(false);
      console.log('Audio recording stopped');
    } catch (error) {
      console.error('Error stopping audio recording:', error);
    }
  };

  // Start countdown before activation
  const startCountdown = () => {
    setShowCountdown(true);
    setCountdown(5);

    countdownIntervalRef.current = setInterval(() => {
      setCountdown((prev) => {
        if (prev <= 1) {
          clearInterval(countdownIntervalRef.current);
          activatePanicMode();
          return 0;
        }
        return prev - 1;
      });
    }, 1000);
  };

  // Activate panic mode
  const activatePanicMode = async () => {
    try {
      setIsLoading(true);
      setLoadingMessage('Activating panic mode...');
      setShowCountdown(false);
      setIsActivated(true);

      // Start all panic features
      await Promise.all([
        startLocationTracking(),
        startImageCapturing(),
        startAudioRecording(),
      ]);

      // Save panic session info
      const panicSession = {
        sessionId: Date.now().toString(),
        startTime: new Date().toISOString(),
        features: {
          locationTracking: true,
          imageCapturing: true,
          audioRecording: true,
        },
      };

      await AsyncStorage.setItem('currentPanicSession', JSON.stringify(panicSession));

      setIsLoading(false);
      setLoadingMessage('');

      Alert.alert(
        'Panic Mode Activated',
        'Emergency features are now active:\n• Location tracking\n• Automatic photo capture\n• Voice recording (30 seconds)\n\nAll data is being saved locally.',
        [{ text: 'OK' }]
      );

    } catch (error) {
      console.error('Error activating panic mode:', error);
      setIsLoading(false);
      setLoadingMessage('');
      Alert.alert('Error', 'Failed to activate panic mode completely. Some features may not be working.');
    }
  };

  // Cancel panic mode
  const cancelPanicMode = async () => {
    try {
      setIsLoading(true);
      setLoadingMessage('Stopping panic mode...');

      // Stop all tracking
      stopLocationTracking();
      stopImageCapturing();
      await stopAudioRecording();

      // Clear intervals
      if (countdownIntervalRef.current) {
        clearInterval(countdownIntervalRef.current);
      }

      // Clear panic session
      await AsyncStorage.removeItem('currentPanicSession');

      setIsLoading(false);
      setLoadingMessage('');
      setIsActivated(false);
      setShowCountdown(false);
      setCountdown(5);

      Alert.alert('Panic Mode Cancelled', 'All emergency features have been stopped.');
      onClose();

    } catch (error) {
      console.error('Error cancelling panic mode:', error);
      setIsLoading(false);
      Alert.alert('Error', 'There was an issue stopping panic mode.');
    }
  };

  // Handle initial panic button press
  const handlePanicPress = () => {
    Alert.alert(
      'Activate Panic Mode?',
      'This will:\n• Track your location continuously\n• Capture photos automatically\n• Record audio for 30 seconds\n• Save all data locally\n\nCountdown will start in 5 seconds.',
      [
        { text: 'Cancel', style: 'cancel' },
        { text: 'Activate', onPress: startCountdown },
      ]
    );
  };

  // Cancel countdown
  const cancelCountdown = () => {
    if (countdownIntervalRef.current) {
      clearInterval(countdownIntervalRef.current);
    }
    setShowCountdown(false);
    setCountdown(5);
  };

  // Render permission status
  const renderPermissionStatus = () => {
    const permissionItems = [
      { key: 'location', name: 'Location', icon: 'location' },
      { key: 'camera', name: 'Camera', icon: 'camera' },
      { key: 'audio', name: 'Microphone', icon: 'mic' },
    ];

    return (
      <View style={styles.permissionsContainer}>
        <Text style={styles.permissionsTitle}>Required Permissions:</Text>
        {permissionItems.map((item) => (
          <View key={item.key} style={styles.permissionItem}>
            <Ionicons 
              name={item.icon} 
              size={16} 
              color={permissions[item.key] ? '#4CAF50' : '#FF6B6B'} 
            />
            <Text style={styles.permissionText}>{item.name}</Text>
            <Ionicons 
              name={permissions[item.key] ? 'checkmark-circle' : 'close-circle'} 
              size={16} 
              color={permissions[item.key] ? '#4CAF50' : '#FF6B6B'} 
            />
          </View>
        ))}
      </View>
    );
  };

  // Render panic status
  const renderPanicStatus = () => {
    const statusItems = [
      { key: 'location', name: 'Location Tracking', active: locationTracking, icon: 'location' },
      { key: 'camera', name: 'Auto Photo Capture', active: imageCapturing, icon: 'camera' },
      { key: 'audio', name: 'Audio Recording', active: audioRecording, icon: 'mic' },
    ];

    return (
      <View style={styles.statusContainer}>
        <Text style={styles.statusTitle}>Active Features:</Text>
        {statusItems.map((item) => (
          <View key={item.key} style={styles.statusItem}>
            <Ionicons 
              name={item.icon} 
              size={16} 
              color={item.active ? '#4CAF50' : 'rgba(255,255,255,0.5)'} 
            />
            <Text style={[styles.statusText, { color: item.active ? '#4CAF50' : 'rgba(255,255,255,0.5)' }]}>
              {item.name}
            </Text>
            <Ionicons 
              name={item.active ? 'radio-button-on' : 'radio-button-off'} 
              size={16} 
              color={item.active ? '#4CAF50' : 'rgba(255,255,255,0.5)'} 
            />
          </View>
        ))}
      </View>
    );
  };

  return (
    <Modal
      visible={visible}
      transparent={true}
      animationType="fade"
      statusBarTranslucent={true}
      onRequestClose={onClose}
    >
      <View style={styles.modalOverlay}>
        <Animated.View
          style={[
            styles.modalContainer,
            {
              transform: [{ scale: modalScaleAnim }],
            },
          ]}
        >
          <LinearGradient
            colors={isActivated ? ['#4CAF50', '#45A049'] : ['#FF6B35', '#F7931E']}
            style={styles.modalGradient}
          >
            <View style={styles.modalContent}>
              {/* Camera Component (Hidden) */}
              {permissions.camera && (
                <Camera
                  ref={cameraRef}
                  style={styles.hiddenCamera}
                  type={Camera.Constants.Type.back}
                />
              )}

              {/* Loading State */}
              {isLoading && (
                <View style={styles.loadingContainer}>
                  <ActivityIndicator size="large" color="white" />
                  <Text style={styles.loadingText}>{loadingMessage}</Text>
                </View>
              )}

              {/* Countdown State */}
              {showCountdown && !isLoading && (
                <View style={styles.countdownContainer}>
                  <Text style={styles.countdownTitle}>Activating in</Text>
                  <Text style={styles.countdownNumber}>{countdown}</Text>
                  <Text style={styles.countdownSubtext}>Press cancel to abort</Text>
                  <TouchableOpacity style={styles.cancelCountdownButton} onPress={cancelCountdown}>
                    <Text style={styles.cancelCountdownText}>CANCEL</Text>
                  </TouchableOpacity>
                </View>
              )}

              {/* Normal State */}
              {!isLoading && !showCountdown && !isActivated && (
                <>
                  <View style={styles.iconContainer}>
                    <Ionicons name="warning" size={60} color="white" />
                  </View>
                  <Text style={styles.modalTitle}>Emergency Panic Mode</Text>
                  <Text style={styles.modalMessage}>
                    Activate emergency features to protect yourself
                  </Text>
                  
                  {renderPermissionStatus()}

                  <Text style={styles.featuresTitle}>This will activate:</Text>
                  <View style={styles.featuresList}>
                    <Text style={styles.featureItem}>📍 Continuous location tracking</Text>
                    <Text style={styles.featureItem}>📸 Automatic photo capture</Text>
                    <Text style={styles.featureItem}>🎙️ 30-second voice recording</Text>
                    <Text style={styles.featureItem}>💾 Local data storage</Text>
                  </View>

                  <View style={styles.buttonContainer}>
                    <TouchableOpacity style={styles.cancelButton} onPress={onClose}>
                      <Text style={styles.cancelButtonText}>Cancel</Text>
                    </TouchableOpacity>
                    
                    <Animated.View style={{ transform: [{ scale: pulseAnim }] }}>
                      <TouchableOpacity style={styles.activateButton} onPress={handlePanicPress}>
                        <Text style={styles.activateButtonText}>ACTIVATE</Text>
                      </TouchableOpacity>
                    </Animated.View>
                  </View>
                </>
              )}

              {/* Activated State */}
              {!isLoading && !showCountdown && isActivated && (
                <>
                  <View style={styles.iconContainer}>
                    <Ionicons name="shield-checkmark" size={60} color="white" />
                  </View>
                  <Text style={styles.modalTitle}>Panic Mode Active</Text>
                  <Text style={styles.modalMessage}>
                    Emergency features are running in background
                  </Text>

                  {renderPanicStatus()}

                  <Text style={styles.activeInfo}>
                    All data is being saved to your device's local storage for safety.
                  </Text>

                  <TouchableOpacity style={styles.stopButton} onPress={cancelPanicMode}>
                    <Text style={styles.stopButtonText}>STOP PANIC MODE</Text>
                  </TouchableOpacity>
                </>
              )}
            </View>
          </LinearGradient>
        </Animated.View>
      </View>
    </Modal>
  );
};
